def factorial(n) {
    result = 1;
    i = n;
    while (i != 0) {
        result = result * i;
        i = i - 1;
    }
    return result;
}

def greet(code) {
    msg = "language code:";
    print(msg);
    print(code);
}

x = 5;
f = factorial(x);
print(f);

pi = 3.14;
r = 3.0;
area = pi *. r *. r;
print(area);

label = "done";

greet(x);
